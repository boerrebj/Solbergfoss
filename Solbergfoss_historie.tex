\begin{frame}
\frametitle{Innbetalingsmodeller}
%

Foreslår fire ulike innbetalingsmodeller:
%
\begin{enumerate}[\Alph*]
\item Betaler beløp \(M_{tot}\) år 0
\item Betaler årlig beløp \(M_{anu}\)
\item Fritak for innbetaling (maks \(n\) år), i påfølgende år innbetales
\(M_{anu2}\) (1)
\item Betaler beløp \(M_{tot}\) + \(M_{ekstra}\) i år 0 (2)
\end{enumerate}

\textbf{Noter:}\\ 
{\small 
\begin{enumerate}
\item Årlig innbetaling av (deler av) rentekostnad 
\item Ekstra innbetaling trekkes fra årsavgift i på følgende år
\end{enumerate}
}
%
\end{frame}


\begin{frame}
\frametitle{Annuitet, fastrentelån}

Årlig innbetaling: 
\[
M_{anu}=C\cdot \frac{r(1+r)^{t}}{(1+r)^{t}-1}
\] 
hvor
\begin{description}
	\item[\(M_{anu}\):] Årlig terminbeløp (avdrag + renter)
	\item[C:] Lånebeløp (kapital)
	\item[r:] Årlig rentesats
	\item[t:] Løpetid i antall år
\end{description}
%
\begin{exampleblock}{Eksempel}
 C= kr 200 000,-, r=6\%, t=23 å, A=kr 16 256,-
 \end{exampleblock}
 %
\end{frame}

\begin{frame}

 \item modell B årlig innbetaling $M_{anu}$ videreføres

\end{itemize}
%
\begin{exampleblock}{Modell A og D}
Løpetid på 23 år, salg etter 10 år, 13*$C_{anu}$ legges til kjøpesummen.
\end{exampleblock}
